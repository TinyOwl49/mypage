```typescript
import p5 from "p5";

type Rect = { x: number; y: number; w: number; h: number };

const sketch = (p: p5) => {
    const width = window.innerWidth / 2;
    const height = window.innerHeight / 2;
    const centerX = width / 2;
    const centerY = height / 2;
    const count = 250;
    const sigmaX = width / 2.5;
    const sigmaY = height / 2.5;
    const minSize = 10;
    const maxSize = Math.min(width, height) / 4;

    let img: p5.Image;
    let mask: p5.Graphics;

    const generateRects = (): Rect[] =>
        Array.from({ length: count }, () => {
            const w = p.random(minSize, maxSize);
            const h = p.random(minSize, maxSize);
            const cx = width - Math.abs(p.randomGaussian(0, sigmaX));
            const cy = height - Math.abs(p.randomGaussian(0, sigmaY));
            return { x: cx - w / 2, y: cy - h / 2, w, h };
        });

    p.setup = async () => {
        p.createCanvas(width, height);
        img = await p.loadImage("./forest.jpeg");
        p.imageMode(p.CENTER);
        mask = p.createGraphics(width, height);
        mask.noStroke();
        mask.fill(255);
        for (const { x, y, w, h } of generateRects()) {
            mask.rect(x, y, w, h);
        }
    };

    p.draw = () => {
        p.blendMode(p.BLEND);
        p.background(255);
        p.image(img, centerX, centerY, width, height);

        p.blendMode(p.EXCLUSION);
        p.image(mask, centerX, centerY);
    };
};
new p5(sketch, document.getElementById("app")!);
```
